/** * 경력인정 계산기 - AI 자동 인식 백엔드 (무료 버전, Google Drive OCR 사용) * * [사용 방법] * 1. script.google.com 에서 "새 프로젝트" 생성 * 2. 이 코드를 전체 복사해서 붙여넣기 * 3. 왼쪽 메뉴 "서비스" 옆 + 버튼 클릭 → "Drive API" 검색해서 추가 * (모양이 "Drive API"라고 나오면 그걸로 추가하면 됩니다. 버전은 기본값 그대로) * 4. 오른쪽 위 "배포" → "새 배포" → 유형: 웹 앱 * - 실행 계정: 나 * - 액세스 권한이 있는 사용자: 전체 * 배포 후 나오는 웹 앱 URL을 계산기 화면의 "⚙ 서버 주소 설정"에 붙여넣으세요. * * [비용] Google 계정 자체의 OCR 기능만 사용하므로 별도 비용이 전혀 들지 않습니다. * [한계] 표 형태의 문서는 완벽하게 구조를 이해하지 못하고, 텍스트 한 줄에서 * "날짜~날짜" 패턴만 찾아 근무처명을 추정합니다. 일부 항목은 놓칠 수 있으니 * 계산기의 "인식 원문 보기"로 원문을 확인해 직접 보완해주세요. */ function doPost(e) { var tempFileIds = []; try { var body = JSON.parse(e.postData.contents); var files = body.files || []; if (files.length === 0) { return jsonOutput({ ok: false, error: '첨부된 파일이 없습니다.' }); } var allText = ''; files.forEach(function (f) { var result = ocrFileToText(f); tempFileIds.push(result.fileId); allText += result.text + '\n'; }); var entries = parseCareerEntries(allText); return jsonOutput({ ok: true, rawText: allText.trim(), entries: entries }); } catch (err) { return jsonOutput({ ok: false, error: String(err) }); } finally { // 임시로 생성된 OCR용 Google 문서 정리 tempFileIds.forEach(function (id) { try { Drive.Files.remove(id); } catch (e2) { /* 무시 */ } }); } } function ocrFileToText(f) { var blob = Utilities.newBlob(Utilities.base64Decode(f.data), f.mimeType, f.name || 'upload'); var resource = { name: (f.name || 'upload') + '_ocr', mimeType: MimeType.GOOGLE_DOCS }; var options = { ocr: true, ocrLanguage: 'ko' }; var docFile = Drive.Files.create(resource, blob, options); var text = ''; var doc = DocumentApp.openById(docFile.id); text = doc.getBody().getText(); return { fileId: docFile.id, text: text }; } /** * OCR 텍스트에서 "YYYY.MM.DD ~ YYYY.MM.DD" 형태의 날짜 범위를 찾아 * 근무경력 항목으로 변환합니다. * * 회사명은 날짜와 같은 줄에 있을 때만 정확히 추출합니다. 표 형태 문서는 * OCR이 열마다 줄 수를 다르게 나눠서, 다른 줄까지 추측하면 오히려 * 엉뚱한 값(퇴사사유 등)이 회사명으로 잘못 들어갈 위험이 있어 일부러 * 추측하지 않습니다. 같은 줄에서 못 찾으면 "(근무처명 확인 필요)"로 * 남겨두니, 계산기의 "인식 원문 보기"를 보고 직접 채워주세요. */ function parseCareerEntries(text) { var rawLines = text.split('\n'); var lines = []; rawLines.forEach(function (l) { var t = l.trim(); if (t.length > 0) lines.push(t); }); var entries = []; // 패턴 1: 숫자+구분기호 (예: 2024.04.01 ~ 2024.12.31 / 2024-04 ~ 24.12) var numericRe = /(\d{2,4})[.\-\/](\d{1,2})(?:[.\-\/](\d{1,2}))?\s*[~\-–—∼〜]\s*(\d{2,4})[.\-\/](\d{1,2})(?:[.\-\/](\d{1,2}))?/; // 패턴 2: 한글 년/월/일 표기 (예: 2024년 4월 ~ 2024년 12월, 2024년 4월 1일 ~ 2024년 12월 31일) var koreanRe = /(\d{4})\s*년\s*(\d{1,2})\s*월(?:\s*(\d{1,2})\s*일)?\s*[~\-–—∼〜]\s*(\d{4})\s*년\s*(\d{1,2})\s*월(?:\s*(\d{1,2})\s*일)?/; // 패턴 3: 시작일만 있고 종료가 "재직중/현재"인 경우 - 숫자 표기 (예: 2024.04 ~ 재직중) var ongoingNumericRe = /(\d{2,4})[.\-\/](\d{1,2})(?:[.\-\/](\d{1,2}))?\s*[~\-–—∼〜]\s*(재직\s*중|현재|근무\s*중)/; // 패턴 4: 시작일만 있고 종료가 "재직중/현재"인 경우 - 한글 표기 (예: 2024년 3월 1일 ~ 재직중) var ongoingKoreanRe = /(\d{4})\s*년\s*(\d{1,2})\s*월(?:\s*(\d{1,2})\s*일)?\s*[~\-–—∼〜]\s*(재직\s*중|현재|근무\s*중)/; lines.forEach(function (line) { var startY, startM, startD, endY, endM, endD, residualSource; var m1 = line.match(numericRe); var m2 = line.match(koreanRe); var m3n = line.match(ongoingNumericRe); var m3k = line.match(ongoingKoreanRe); if (m3k) { startY = m3k[1]; startM = pad2(m3k[2]); startD = m3k[3] ? pad2(m3k[3]) : '01'; var today1 = new Date(); endY = String(today1.getFullYear()); endM = pad2(today1.getMonth() + 1); endD = pad2(today1.getDate()); residualSource = line.replace(ongoingKoreanRe, ''); } else if (m3n) { startY = normalizeYear(m3n[1]); startM = pad2(m3n[2]); startD = m3n[3] ? pad2(m3n[3]) : '01'; var today2 = new Date(); endY = String(today2.getFullYear()); endM = pad2(today2.getMonth() + 1); endD = pad2(today2.getDate()); residualSource = line.replace(ongoingNumericRe, ''); } else if (m2) { startY = m2[1]; startM = pad2(m2[2]); startD = m2[3] ? pad2(m2[3]) : '01'; endY = m2[4]; endM = pad2(m2[5]); endD = m2[6] ? pad2(m2[6]) : lastDayOfMonth(m2[4], pad2(m2[5])); residualSource = line.replace(koreanRe, ''); } else if (m1) { startY = normalizeYear(m1[1]); startM = pad2(m1[2]); startD = m1[3] ? pad2(m1[3]) : '01'; endY = normalizeYear(m1[4]); endM = pad2(m1[5]); endD = m1[6] ? pad2(m1[6]) : lastDayOfMonth(endY, endM); residualSource = line.replace(numericRe, ''); } else { return; } var residual = cleanNameCandidate(residualSource); var name = isPlausibleName(residual) ? residual : ''; entries.push({ name: name || '(근무처명 확인 필요)', start: startY + '-' + startM + '-' + startD, end: endY + '-' + endM + '-' + endD }); }); return entries; } // 괄호 부가설명(예: (2년이상), (만2년), (1년이상)) 및 앞뒤 구두점 제거 function cleanNameCandidate(s) { s = s.replace(/\([^\)]{1,12}\)/g, '').trim(); s = s.replace(/^[\s|·・:\-]+/, '').replace(/[\s|·・:\-]+$/, ''); return s; } // 한글이 포함된 최소 2자 이상 텍스트만 회사명 후보로 인정 function isPlausibleName(s) { if (!s) return false; if (s.length < 2) return false; if (!/[가-힣]/.test(s)) return false; return true; } function normalizeYear(y) { y = String(y); if (y.length === 2) { var n = Number(y); return (n > 50 ? '19' : '20') + y; // 2자리 연도는 20xx로 추정 (50 초과 시 19xx) } return y; } function pad2(n) { return String(n).length === 1 ? '0' + n : String(n); } function lastDayOfMonth(y, m) { var d = new Date(Number(y), Number(m), 0).getDate(); return pad2(d); } function jsonOutput(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /** * 배포 후 브라우저로 GET 접속했을 때 정상 동작 확인용 (선택사항) */ function doGet() { return ContentService .createTextOutput(JSON.stringify({ ok: true, message: '경력계산기 OCR 인식 서버가 정상 동작 중입니다.' })) .setMimeType(ContentService.MimeType.JSON); }
